\question[10]\mbox{}
\textbf{TRUE or FALSE!}  Write  TRUE if the statement is true.  Otherwise, write FALSE.  Your response should be in ALL CAPS.  No justification is required.
\begin{enumerate}[(a)]
\item a monotone increasing sequence must converge 
\vspace{1.2in}
\item in any metric space a closed and bounded set must be compact
\vspace{1.2in}
\item if a set is not closed, then it must be open
\vspace{1.2in}
\item in a metric space with the discrete metric, every set is both open and closed
\vspace{1.2in}
\item if $\lim_{x\rightarrow 0} f(x) = 4$ then $f(0) = 4$
\vspace{1.2in}
\end{enumerate}

\newpage
\question[10]\mbox{}

\begin{enumerate}[(a)]
\item
Write down the definition of a metric space.
\vspace{3in}
\item 
Write down three different examples of metrics we can define on $\mathbb{R}^2$. (Not just the names!)
\vspace{2in}
\item 
Explain why $d(x,y) = (x-y)^2$ is NOT a metric on $\mathbb{R}$.
\end{enumerate}

\newpage
\question[10]\mbox{}
\begin{enumerate}[(a)]
\item Write down the definition of a sequence $\{x_n\}$ of values in a metric space $(M,d)$ converging to a value $L\in M$.
\vspace{1in}

\item Prove that if a sequence $\{x_n\}$ converges to a value $L\in M$, then there exists a constant $R>0$ such that $d(x_n,L) < R$ for all $n\in\mathbb{Z}_+$ without using any theorems from class.
\end{enumerate}

\newpage
\question[10]\mbox{}

\begin{enumerate}[(a)]
\item
Write down the definition of a sequence $\{x_n\}$ in a metric space $(M,d)$ being Cauchy
\vspace{1in}
\item
Give an example of a metric space with a Cauchy sequence that does not converge.
\vspace{2in}
\item 
Suppose that $M=\mathbb R$ and $d$ is the discrete metric.  Prove that if $\{x_n\}$ is a Cauchy sequence, then the set
$$X = \{x_1,x_2,x_3,\dots\}$$
is finite.
\end{enumerate}

\newpage
\question[10]\mbox{}
Carefully state each of the following theorems.
\begin{enumerate}[(a)]
\item the Lindel\"{o}f Covering Theorem
\vspace{2in}
\item the Bolzano-Weierstrass Theorem
\vspace{2in}
\item the Cantor Intersection Theorem
\vspace{2in}
\item the Heine-Borel Theorem
\end{enumerate}

\newpage
\question[10]\mbox{}

\begin{enumerate}[(a)]
\item  Let $(S,d_S)$ and $(T,d_T)$ be metric spaces and $f: S\rightarrow T$ a function, and suppose $a\in S$.
Write the definition of $\lim_{x\rightarrow a} f(x) = L$.
\vspace{1in}
\item  Consider the function $f:\mathbb{R}\rightarrow\mathbb{R}$ defined by
$$f(x) = \left\lbrace\begin{array}{cc}
1, & x\in\mathbb{Q}\\
0, & x\notin\mathbb{Q}
\end{array}\right.$$
Prove that the limit $\lim_{x\rightarrow 0} f(x)$ does not exist.
\vspace{2.5in}
\item  Consider the function $f:[0,\infty)\rightarrow\mathbb{R}$ defined by $f(x) = \sqrt{x}$.
Prove that $\lim_{x\rightarrow 4} f(x) = 2$ using the definition of a limit.

\end{enumerate}

\newpage
\question[10]\mbox{}
\begin{enumerate}[(a)]
\item Write down the definition of a subset $A\subseteq M$ of a metric space $M$ being compact.
\vspace{2in}
\item Prove that if $A = \{a\}$ for some $a\in M$, then $A$ is compact.
\end{enumerate}
