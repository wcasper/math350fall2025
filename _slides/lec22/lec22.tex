\section{Real Analysis Lecture 22}

\subsection{Riemann-Stieltjes integrals}

\begin{frame}{Integration by parts}
The integrand and integrator play dual roles:
\pause
\begin{thm}{Integration by parts}
\pause
Let $f$ be integrable with respect to $\alpha$ on $[a,b]$.\\
\pause
Then $\alpha$ is integrable with respect to $f$ on $[a,b]$ and
$$\int_a^b f d\alpha = f(b)\alpha(b)-f(a)\alpha(a)-\int_a^b \alpha df.$$
\end{thm}
\end{frame}

\begin{frame}{Integration by parts}
\begin{proof}
\pause
Let $P = \{x_0,\dots, x_n\}$ be a partition of $[a,b]$.\\
\pause
The starting place of the proof is the telescoping sum
$$f(b)\alpha(b)-f(a)\alpha(a) = \sum_{k=1}^n \left[f(x_k)\alpha(x_k)-f(x_{k-1})\alpha(x_{k-1})\right].$$
\pause
Then the left-hand side of the equality in the theorem motivates
{\small
\begin{align*}
f(b)&\alpha(b)-f(a)\alpha(a)-S(P,\alpha,f,\{t_k\})\\
  &= \sum_{k=1}^n\left[f(x_k)\alpha(x_k)-f(x_{k-1})\alpha(x_{k-1})-\alpha(t_k)f(x_k)+\alpha(t_k)f(x_{k-1})\right].
\end{align*}
}
\end{proof}
\end{frame}

\begin{frame}{Integration by parts}
\begin{proof}
If we think about the partition:
$$P' = P\cup\{t_k\} = \{x_0,t_1,x_1,t_2,x_2,\dots, t_n,x_n\}$$
\pause
and the sample points:
$$t_1'=x_0,\ t_2'=x_1,\ t_3'=x_1,\ t_4'=x_2,\ \dots,\ t_{2n}'=x_n,\ t_{2n+1}'=x_n$$
\pause
We can rewrite this as
$$f(b)\alpha(b)-f(a)\alpha(a)-S(P,f,\alpha,\{t_k'\}) = S(P,\alpha,f,\{t_k\}).$$
\end{proof}
\end{frame}

\begin{frame}{Integration by parts}
\begin{proof}
\pause
Now let $\epsilon > 0$ and set $A = f(b)\alpha(b)-f(a)\alpha(a)-\int_a^bf d\alpha$.\\
\pause
Choose a partition $P_\epsilon$ such that for any refinement $P$ of $P_\epsilon$
$$\left| S(P,f,\alpha,\{t_k\})-\int_a^b fd\alpha \right| < \epsilon$$
for all choices of sample points.\\
\pause
Let $P$ be a refinement of $P_\epsilon$ and $\{t_k\}$ be a choice of sample points.\\
\pause
Then for $P'$ and $\{t_k'\}$ defined as above, $P'$ is a refinement of $P_\epsilon$, so
\begin{align*}
|S(P,\alpha,f,\{t_k\})-A|
&= \left|S(P',f,\alpha,\{t_k'\})-\int_a^bf d\alpha\right| < \epsilon
\end{align*}
\end{proof}
\end{frame}

\begin{frame}{Uniform continuity}
\begin{defn}
Let $(S,d_S)$ and $(T,d_T)$ are metric spaces, $A\subseteq S$, and $f: A\rightarrow T$.\\
\pause
Then we say $f$ is \textbf{uniformly continuous} on $A$ if for all $\epsilon > 0$ there exists $\delta > 0$ such that
$$d_S(x,y) < \delta\ \ \Rightarrow\ \ d_T(f(x),f(y)) < \epsilon\ \forall x,y\in A.$$
\end{defn}
\begin{itemize}
\pause
\item uniform continuity implies continuity
\pause
\item uniform continuity = same $\delta$ works for all $c$
\pause
\item continuity implies uniform continuity on \emph{compact sets}
\end{itemize}
\end{frame}

\begin{frame}{Uniform continuity}
\begin{thm}
Let $(S,d_S)$ and $(T,d_T)$ are metric spaces, $A\subseteq S$, and $f: A\rightarrow T$.\\
If $f$ is continuous on $A$ and $A$ is compact, then $f$ is uniformly continuous on $A$.
\end{thm}
\pause
\begin{proof}
\pause
Assume continuity and compactness, and let $\epsilon > 0$.\\
\pause
Then for all $c\in A$, there exists $\delta_c > 0$ with
$$d_S(x,c) < \delta_c\ \ \Rightarrow\ \ d_T(f(x),f(c)) < \epsilon/2\ \forall x,y\in A.$$
\pause
Since $A$ is compact, the open cover $\{B_S(c,\delta_c/2): c\in A\}$ of $A$ has a finite subcover
$\{B_S(c_1,\delta_{c_1}/2),\dots,B_S(c_r,\delta_{c_r}/2)\}$
\end{proof}
\end{frame}

\begin{frame}{Uniform continuity}
\begin{proof}
\pause
Take $\delta = \min\{\delta_{c_1},\delta_{c_2},\dots,\delta_{c_r}\}/2$.\\
\pause
Suppose $x,y\in A$ with $d_S(x,y) < \delta$.\\
\pause
Then $x\in B_S(c_j,\delta_{c_j}/2)$ for some $j$, and therefore
$$d_S(c_j,y)\leq d_S(c_j,x)+d_S(x,y) < \delta_{c_1}/2 + \delta < \delta_{c_j}.$$
\pause
This means that $x,y\in B_S(c_j,\delta_{c_j})$, and therefore
$$d_T(f(x),f(y)) \leq d_T(f(x),c_j) + d_T(c_j,f(y) < \epsilon/2+\epsilon/2=\epsilon.$$
\pause
Since $x,y\in A$ and $\epsilon > 0$ was arbitrary, this proves uniform continuity.
\end{proof}
\end{frame}

\begin{frame}{Relation to Riemann integrals}
\begin{thm}
Let $f$ be integrable with respect to $\alpha$ on $[a,b]$.\\
If $\alpha$ is continuous on $[a,b]$ and continuously differentiable on $(a,b)$, then
$f(x)\alpha'(x)$ is integrable with respect to $x$ on $[a,b]$ and
$$\int_a^b f d\alpha = \int_a^bf(x)\alpha'(x)dx.$$
\end{thm}
\end{frame}

\begin{frame}{Relation to Riemann integrals}
\begin{proof}
\pause
Set $M=\max \{|\alpha'(x)|: a\leq x \leq b\}$ and let $\epsilon > 0$.\\
\pause
Since $\alpha'(x)$ is continuous on $[a,b]$, it is uniformly continuous on $[a,b]$.\\
\pause
Therefore there exists $\delta > 0$ with
$$|x-y|<\delta\ \ \Rightarrow\ \ |\alpha'(x)-\alpha'(y)| < \epsilon/2M(b-a),\ \forall x,y\in [a,b].$$
\pause
Choose $N\in\mathbb{Z}_+$ with $N>(b-a)/\epsilon$ so that $(b-a)/N < \epsilon$\\
\pause
Let $P_{\epsilon} = \{x_0,x_1,\dots, x_n\}$ with $x_j = (b-a)j/N + a$.\\
\end{proof}
\end{frame}


\begin{frame}{Relation to Riemann integrals}
\begin{proof}
Then for any partition $P$ finer than $P_\epsilon$, the Mean Value Theorem says we can choose $\{\tau_k\}$ with
$$f(x_k)-f(x_{k-1})=f'(\tau_k)(x_k-x_{k-1}),$$
\pause
so that
$$S(P,f,\alpha,\{\tau_k\}) = S(P,f\alpha',x,\{\tau_k\})$$
\pause
Therefore for any sample points $t_k$
$$S(P,f\alpha',x,\{t_k\}) - S(P,f,\alpha,\{\tau_k\}) < \epsilon /2.$$
\end{proof}
\end{frame}

\begin{frame}{Relation to Riemann integrals}
\begin{proof}
We may also choose $P_\epsilon'$ such that for any refinement $P$ of $P_\epsilon'$, we have
$$\left|S(P,f,\alpha,\{t_k\}) - \int_a^bfd\alpha\right| < \epsilon/2.$$
\pause
Therefore, for $P_\epsilon''=P_\epsilon\cup P_\epsilon'$, and any refinement $P$ of $P_\epsilon''$:
\pause
\begin{align*}
\left|S(P,f\alpha',x,\{t_k\}) - \int_a^bfd\alpha\right| 
&\leq      |S(P,f\alpha',x,\{t_k\})-S(P,f,\alpha,\{\tau_k\})|\\
&+    \left|S(P,f,\alpha,\{\tau_k\})-\int_a^b fd\alpha\right| < \epsilon.
\end{align*}
\end{proof}
\end{frame}

\subsection{Stieltjes integrals}

\begin{frame}{Concrete Example}
Let $f(x) = x$ and $\alpha(x) =x$, and consider $\int_0^1 f d\alpha$.\\
\pause
If we consider the partition
$$P = \{x_0=0/n,x_1=1/n,\dots,x_n=n/n\},$$
then for left or right sample points $t_k=x_{k-1}$,
$$S(P,f,\alpha,\{x_{k-1}\}) = \sum_{k=1}^n x_{k-1}(x_k-x_{k-1}) = \sum_{k=1}^n \frac{k-1}{n^2} = \frac{(n-1)}{2n}$$
$$S(P,f,\alpha,\{x_{k}\}) = \sum_{k=1}^n x_{k}(x_k-x_{k-1}) = \sum_{k=1}^n \frac{k}{n^2} = \frac{(n+1)}{2n}$$
In either case the limit goes to $1/2$.
\end{frame}

\begin{frame}{Concrete Example}
Guess: $\int_0^1 fd\alpha = 1/2$.
\pause
What about other sample points?
$$S(P,f,\alpha,\{t_k\} = \sum_{k=1}^n f(t_k)\Delta \alpha_k(P) = \sum_{k=1}^n t_k(x_k-x_{k-1}).$$
\pause
Since $x$ is monotone increasing:
{\small
$$\frac{n-1}{2n}= \sum_{k=1}^n x_{k-1}(x_k-x_{k-1})\leq S(P,f,\alpha,\{t_k\}) \leq \sum_{k=1}^n x_k(x_k-x_{k-1}) = \frac{n+1}{2n}.$$
}
$$\text{\color{red}What about other partitions, though?}$$
\end{frame}

\begin{frame}{Stieltjes sums}
Let $f$ and $\alpha$ be real-valued functions on $[a,b]$, and let $P=\{x_0,x_1,\dots,x_n\}$ be a partition of $[a,b]$ and define\\
{\small
$$m_k(f) = \inf\{f(x): x_{k-1}\leq x\leq x_k\},\ \ M_k(f) = \sup\{f(x): x_{k-1}\leq x\leq x_k\}.$$
}
\pause
The \textbf{upper Stieltjes sum} of $f$ with respect to $\alpha$ on $[a,b]$ is:
$$\overline{S}(P,f,\alpha) = \sum_{k=1}^n M_k(f)\Delta\alpha_k(P),$$
\pause
and the \textbf{lower Stieltjes sum} of $f$ with respect to $\alpha$ on $[a,b]$ is:
$$\underline{S}(P,f,\alpha) = \sum_{k=1}^n m_k(f)\Delta\alpha_k(P)$$
\end{frame}

\begin{frame}{Stieltjes sums}
These sums refine predictably!
\begin{lem}
Let $f$ and $\alpha$ be real-valued functions on $[a,b]$, with $\alpha$ monotone increasing.
\pause
If $P$ is a partition of $[a,b]$ and $P'$ is a refinement of $P$, then
$$
\underline{S}(P,f,\alpha) \leq
\underline{S}(P',f,\alpha) \leq
\overline{S}(P',f,\alpha) \leq
\overline{S}(P,f,\alpha)
$$
\end{lem}
\end{frame}

\begin{frame}{Stieltjes sums}
\begin{proof}
The inequality $\underline{S}(P',f,\alpha) \leq \overline{S}(P',f,\alpha)$ is clear.\\
\pause
We will prove $\underline{S}(P,f,\alpha) \leq \underline{S}(P',f,\alpha)$.\\
\pause
Let $P=\{x_0,\dots, x_n\}$ and $P'=\{x_0',\dots, x_N'\}$.\\
\pause
For each $k$, let $\iota(k)$ with $x_{\iota(k)-1} \leq x_k'\leq x_{\iota(k)}$.\\
\pause
{
\tiny
$$m_{\iota(k)}(f)=\inf\{f(x): x_{\iota(k)-1}\leq x\leq x_{\iota(k)}\}\leq \inf\{f(x): x_{k-1}'\leq x\leq x_k'\}=m_k'(f)$$
}
\pause
Therefore 
{\tiny
\begin{align*}
\underline{S}(P',f,\alpha)
 &=    \sum_{k=1}^N m_k'(f)\Delta\alpha_k(P')\\
 &\geq \sum_{k=1}^N m_{\iota(k)}(f)\Delta\alpha_k(P')\\
 & = \sum_{k=1}^n m_{k}(f)\Delta\alpha_k(P) = \underline{S}(P,f,\alpha)
\end{align*}
}
\end{proof}
\end{frame}

\begin{frame}{Stieltjes integrals}
Let $f$ and $\alpha$ be real-valued functions on $[a,b]$, with $\alpha$ monotone increasing.
\pause
The \textbf{upper Stieltjes integral} of $f$ with respect to $\alpha$ on $[a,b]$ is:
$$\overline{\int} fd\alpha = \inf\{\overline{S}(P,f,\alpha): P\ \text{partition of}\ [a,b]\}$$
\pause
The \textbf{lower Stieltjes integral} of $f$ with respect to $\alpha$ on $[a,b]$ is:
$$\underline{\int} fd\alpha = \sup\{\underline{S}(P,f,\alpha): P\ \text{partition of}\ [a,b]\}$$
\end{frame}

\begin{frame}{Riemann's criterion}
\begin{defn}
Let $f$ and $\alpha$ be real-valued functions on $[a,b]$.\\
Then $f$ \textbf{satisfies Riemann's criterion} with respect to $\alpha$ on $[a,b]$ if for all $\epsilon > 0$ there exists a partition $P_\epsilon$ such that
$$0\leq \overline S(P,f,\alpha)-\underline S(P,f,\alpha) < \epsilon.$$
\end{defn}
\pause
\begin{thm}
Let $f$ and $\alpha$ be real-valued functions on $[a,b]$, with $\alpha$ monotone increasing.  Then the following are equivalent:\\
\pause
(a) $f$ is Riemann integrable with respect to $\alpha$ on $[a,b]$\\
\pause
(b) $f$ satisfies Riemann's criterion with respect to $\alpha$ on $[a,b]$\\
\pause
(c) $\underline I(f,\alpha) = \overline I(f,\alpha)$
\end{thm}
\end{frame}
